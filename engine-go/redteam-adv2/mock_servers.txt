# -*- coding: utf-8 -*-
"""对抗测试本地靶站（零外网，全部绑定本机/宿主内网网卡）。

用法: PYTHONUTF8=1 python mock_servers.txt <A端口> <B端口> [B绑定IP]

  A 服务器（入口靶站，127.0.0.1）:
    /r302/*     -> 302 跳转到 http://<B绑定IP>:<B端口>/land?orig=<原路径>   （重定向打进内网段的载体）
    /loop302/*  -> 302 跳到自己（重定向环，测 10 跳上限）
    /real/*     -> mockweb 风格真阳性（swagger/actuator/.env/robots）
    /soft404/*  -> 万物 200 SPA 外壳（catch-all）
    /jump/*     -> 302 跳到 B（单跳，非全站）
    其它        -> 404
  B 服务器（内网段着陆靶标，绑定 RFC1918 地址 192.168.88.1）:
    任何请求 -> 200，body 含全部探测标记 + orig 回显（保证每个端点响应形态不同，
                避开引擎的 catch-all 基线过滤），并把每次命中写入 land_log.txt。

仅用于授权范围内的本地对抗测试。
"""
import sys
import threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from urllib.parse import quote

A_PORT = int(sys.argv[1]) if len(sys.argv) > 1 else 18081
B_PORT = int(sys.argv[2]) if len(sys.argv) > 2 else 18082
B_BIND = sys.argv[3] if len(sys.argv) > 3 else "192.168.88.1"
A_LOG = "a_log.txt"
B_LOG = "land_log.txt"

# 全部 27 个端点的判定标记（apiunauth.Endpoints 的 Expect 并集，小写）
ALL_MARKERS = (
    'swagger-ui "swagger" "openapi" "paths" redoc "_links" "href" "propertysources" '
    '"status" "mappings" "beans" java profile druid "result" "errors" "data" graphiql '
    '<applications nacos username "pageitems" "member" h2-console pprof "# help" email'
)

SPA = "<!doctype html><html><head><title>My App</title></head>\n<body><div id=\"app\"></div></body></html>"

REAL = {
    "/swagger-ui.html": ("text/html", "<html><head><title>Swagger UI</title></head>"
                                     "<body><div id=\"swagger-ui\"></div><script>SwaggerUIBundle()</script></body></html>"),
    "/v3/api-docs": ("application/json", '{"openapi":"3.0.1","info":{"title":"Demo API"},"paths":{"/user":{"get":{}}}}'),
    "/actuator": ("application/json", '{"_links":{"self":{"href":"http://x/actuator"},"env":{"href":"http://x/actuator/env"}}}'),
    "/actuator/env": ("application/json", '{"activeProfiles":["prod"],"propertySources":[{"name":"c","properties":{"db.pw":{"value":"Sup3rS3cret"}}}]}'),
    "/actuator/heapdump": ("application/octet-stream", "JAVA PROFILE 1.0.8\x00\x00\x00"),
    "/.env": ("text/plain", "APP_ENV=production\nDB_PASSWORD=Sup3rS3cret\n"),
    "/robots.txt": ("text/plain", "User-agent: *\nDisallow: /admin\n"),
}


def log(fname, line):
    with open(fname, "a", encoding="utf-8") as f:
        f.write(line + "\n")


class AHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        log(A_LOG, "GET %s | UA=%s" % (self.path, self.headers.get("User-Agent", "")))
        parts = self.path.split("?")[0].strip("/").split("/")
        scene = parts[0] if parts and parts[0] else ""
        sub = "/" + "/".join(parts[1:]) if len(parts) > 1 else "/"
        if scene == "r302" or scene == "jump":
            orig = quote(self.path, safe="")
            loc = "http://%s:%d/land?orig=%s&from=A" % (B_BIND, B_PORT, orig)
            self.send_response(302)
            self.send_header("Location", loc)
            self.send_header("Content-Length", "0")
            self.end_headers()
        elif scene == "loop302":
            self.send_response(302)
            self.send_header("Location", "/loop302/self")
            self.send_header("Content-Length", "0")
            self.end_headers()
        elif scene == "real":
            e = REAL.get(sub)
            if e:
                body = e[1].encode("utf-8")
                self.send_response(200)
                self.send_header("Content-Type", e[0])
                self.send_header("Content-Length", str(len(body)))
                self.end_headers()
                self.wfile.write(body)
            else:
                body = b"<html><body>404</body></html>"
                self.send_response(404)
                self.send_header("Content-Length", str(len(body)))
                self.end_headers()
                self.wfile.write(body)
        elif scene == "soft404":
            body = SPA.encode("utf-8")
            self.send_response(200)
            self.send_header("Content-Type", "text/html; charset=utf-8")
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)
        else:
            body = b"<html><body>404</body></html>"
            self.send_response(404)
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)

    def log_message(self, *a):
        pass


class BHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        # orig= 参数回显进 body：每个端点形态不同，绕开基线过滤，证明"逐端点"着陆
        orig = ""
        for kv in self.path.split("?")[1].split("&") if "?" in self.path else []:
            if kv.startswith("orig="):
                orig = kv[5:]
        body = ("<html><head><title>INTRANET-LANDING</title></head><body>"
                "INTRANET-LANDED orig=%s %s</body></html>" % (orig, ALL_MARKERS)).encode("utf-8")
        log(B_LOG, "HIT %s | UA=%s" % (self.path, self.headers.get("User-Agent", "")))
        self.send_response(200)
        self.send_header("Content-Type", "text/html; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, *a):
        pass


def main():
    a = ThreadingHTTPServer(("127.0.0.1", A_PORT), AHandler)
    b = ThreadingHTTPServer((B_BIND, B_PORT), BHandler)
    log(A_LOG, "=== A up on 127.0.0.1:%d ===" % A_PORT)
    log(B_LOG, "=== B up on %s:%d ===" % (B_BIND, B_PORT))
    ta = threading.Thread(target=a.serve_forever, daemon=True)
    tb = threading.Thread(target=b.serve_forever, daemon=True)
    ta.start()
    tb.start()
    print("A=http://127.0.0.1:%d  B=http://%s:%d  (Ctrl+C 退出)" % (A_PORT, B_BIND, B_PORT))
    try:
        ta.join()
        tb.join()
    except KeyboardInterrupt:
        pass


if __name__ == "__main__":
    main()
